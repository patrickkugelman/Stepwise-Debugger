function(stepwise_add_layer name)
    cmake_parse_arguments(PARSE_ARGV 1 LAYER "" "" "SOURCES;DEPS")
    add_library(stepwise_${name} STATIC ${LAYER_SOURCES})
    add_library(stepwise::${name} ALIAS stepwise_${name})
    target_include_directories(stepwise_${name} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
    target_link_libraries(stepwise_${name} PUBLIC ${LAYER_DEPS} PRIVATE stepwise_warnings)
    set_target_properties(stepwise_${name} PROPERTIES POSITION_INDEPENDENT_CODE ON)
endfunction()
